// AUTOMATED DATA UPDATE ENGINE (FINNHUB API INTEGRATION) async function autoUpdateData() { const btn = document.getElementById('fetchBtn'); const icon = document.getElementById('fetchIcon'); // 1. PASTE YOUR API KEY HERE: const apiKey = 'YOUR_FINNHUB_API_KEY_HERE'; // Get today's date in YYYY-MM-DD format const today = new Date().toISOString().split('T')[0]; const url = `https://finnhub.io/api/v1/calendar/economic?from=${today}&to=${today}&token=${apiKey}`; btn.disabled = true; btn.classList.add('opacity-75'); icon.className = "fa-solid fa-spinner fa-spin"; try { // Fetch live economic calendar data from Finnhub const res = await fetch(url); const data = await res.json(); if (data && data.economicCalendar) { console.log("Fetched Live Releases:", data.economicCalendar); // Loop through live releases and match them to your app's indicators data.economicCalendar.forEach(release => { for (const currency in appData) { appData[currency].forEach(ind => { // Match by indicator name (e.g., "CPI", "Unemployment Rate", "PMI") if (ind.name.toLowerCase().includes(release.event.toLowerCase()) && release.actual !== null) { ind.actual = parseFloat(release.actual); if (release.prev !== null) ind.previous = parseFloat(release.prev); if (release.estimate !== null) ind.forecast = parseFloat(release.estimate); } }); } }); } const now = new Date(); const timeString = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' }); document.getElementById('lastUpdatedNotice').innerText = `Last updated: Today at ${timeString}`; // Recalculate totals and refresh the charts/bars refreshApp(); } catch (err) { alert("API Fetch Error: " + err.message); } finally { btn.disabled = false; btn.classList.remove('opacity-75'); icon.className = "fa-solid fa-cloud-arrow-down"; } }
